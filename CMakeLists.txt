cmake_minimum_required(VERSION 3.20)
project(omni_native_core_prototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(omni_native_core STATIC native_core/omni_native_core.cpp)
target_include_directories(omni_native_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native_core")
add_executable(test_omni_native_core tests/test_core.cpp)
target_link_libraries(test_omni_native_core PRIVATE omni_native_core)
# Release CI must execute the assertions in this test executable.
if(MSVC)
    target_compile_options(test_omni_native_core PRIVATE /UNDEBUG)
else()
    target_compile_options(test_omni_native_core PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME native_core_tests COMMAND test_omni_native_core)
